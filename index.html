ContractTrack.sol

Deploy once. Projects call registerContract() once, then push activity via logUser() and logTx(). Anyone queries with the view functions.

// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

/// @title  ContractTrack
/// @notice Public analytics registry. Projects register their contracts,
///         then push user and tx activity onchain. Anyone can query.
/// @dev    Deploy once; no ownership required for logging — any caller
///         can push data for a registered contract.
contract ContractTrack {

    // ─── Data structures ───────────────────────────────────────

    struct ContractInfo {
        address contractAddr;
        string  name;
        string  category;
        string  description;
        address registeredBy;
        uint256 registeredAt;
        bool    exists;
    }

    struct UserRecord {
        address wallet;
        string  eventType;   // e.g. "Registration", "FirstTx"
        uint256 firstSeenAt;
        uint256 txCount;      // txs this wallet made on this contract
    }

    struct TxRecord {
        address wallet;
        string  action;
        uint256 timestamp;
        uint256 blockNumber;
    }

    struct Stats {
        uint256 totalUsers;   // all logUser() calls
        uint256 uniqueWallets;// distinct wallet addresses
        uint256 totalTxs;     // all logTx() calls
        uint256 eventCount;   // all events emitted
    }

    // ─── Storage ───────────────────────────────────────────────

    // contractAddr => metadata
    mapping(address => ContractInfo) public contracts;

    // contractAddr => stats
    mapping(address => Stats) public stats;

    // contractAddr => wallet => UserRecord
    mapping(address => mapping(address => UserRecord)) public userRecords;

    // contractAddr => list of unique wallet addresses
    mapping(address => address[]) public contractWallets;

    // contractAddr => tx history
    mapping(address => TxRecord[]) public txHistory;

    // all registered contract addresses
    address[] public registeredContracts;

    // ─── Events ────────────────────────────────────────────────

    event ContractRegistered(
        address indexed contractAddr,
        string name,
        address registeredBy
    );
    event UserLogged(
        address indexed contractAddr,
        address indexed wallet,
        string eventType,
        bool isNew
    );
    event TxLogged(
        address indexed contractAddr,
        address indexed wallet,
        string action,
        uint256 blockNumber
    );

    // ─── Registration ──────────────────────────────────────────

    function registerContract(
        address contractAddr,
        string calldata name,
        string calldata category,
        string calldata description
    ) external {
        require(!contracts[contractAddr].exists, "Already registered");
        require(bytes(name).length > 0, "Name required");

        contracts[contractAddr] = ContractInfo({
            contractAddr: contractAddr,
            name: name,
            category: category,
            description: description,
            registeredBy: msg.sender,
            registeredAt: block.timestamp,
            exists: true
        });

        registeredContracts.push(contractAddr);
        emit ContractRegistered(contractAddr, name, msg.sender);
    }

    // ─── Activity logging ──────────────────────────────────────

    /// @notice Log a user interaction. Increments uniqueWallets only on first call per wallet.
    function logUser(
        address contractAddr,
        address wallet,
        string calldata eventType
    ) external {
        require(contracts[contractAddr].exists, "Contract not registered");

        Stats storage s = stats[contractAddr];
        UserRecord storage u = userRecords[contractAddr][wallet];

        bool isNew = u.firstSeenAt == 0;
        if (isNew) {
            userRecords[contractAddr][wallet] = UserRecord({
                wallet: wallet,
                eventType: eventType,
                firstSeenAt: block.timestamp,
                txCount: 1
            });
            contractWallets[contractAddr].push(wallet);
            s.uniqueWallets++;
        } else {
            u.txCount++;
        }

        s.totalUsers++;
        s.eventCount++;

        emit UserLogged(contractAddr, wallet, eventType, isNew);
    }

    /// @notice Log a transaction from a wallet against a contract.
    function logTx(
        address contractAddr,
        address wallet,
        string calldata action
    ) external {
        require(contracts[contractAddr].exists, "Contract not registered");

        txHistory[contractAddr].push(TxRecord({
            wallet: wallet,
            action: action,
            timestamp: block.timestamp,
            blockNumber: block.number
        }));

        stats[contractAddr].totalTxs++;
        stats[contractAddr].eventCount++;

        // bump per-wallet tx count if user is known
        if (userRecords[contractAddr][wallet].firstSeenAt != 0) {
            userRecords[contractAddr][wallet].txCount++;
        }

        emit TxLogged(contractAddr, wallet, action, block.number);
    }

    // ─── Query functions (all free, public) ────────────────────

    /// @notice Get full stats for a contract in one call.
    function getStats(address contractAddr)
        external view
        returns (
            uint256 totalUsers,
            uint256 uniqueWallets,
            uint256 totalTxs,
            uint256 eventCount
        )
    {
        Stats memory s = stats[contractAddr];
        return (s.totalUsers, s.uniqueWallets, s.totalTxs, s.eventCount);
    }

    /// @notice Check if a contract has been used at all.
    function hasUsers(address contractAddr) external view returns (bool) {
        return stats[contractAddr].uniqueWallets > 0;
    }

    /// @notice Get all unique wallet addresses for a contract.
    function getWallets(address contractAddr)
        external view returns (address[] memory)
    {
        return contractWallets[contractAddr];
    }

    /// @notice Get the per-wallet tx count for a specific address.
    function getWalletTxCount(address contractAddr, address wallet)
        external view returns (uint256)
    {
        return userRecords[contractAddr][wallet].txCount;
    }

    /// @notice Get paginated tx history for a contract.
    function getTxHistory(
        address contractAddr,
        uint256 offset,
        uint256 limit
    ) external view returns (TxRecord[] memory page) {
        TxRecord[] storage all = txHistory[contractAddr];
        uint256 end = offset + limit > all.length ? all.length : offset + limit;
        page = new TxRecord[](end - offset);
        for (uint256 i = offset; i < end; i++) {
            page[i - offset] = all[i];
        }
    }

    /// @notice Get all registered contract addresses.
    function getAllContracts() external view returns (address[] memory) {
        return registeredContracts;
    }

    /// @notice Get metadata for a contract.
    function getContractInfo(address contractAddr)
        external view returns (ContractInfo memory)
    {
        return contracts[contractAddr];
    }
}
